async function handleGenerate() { const promptInput = document.querySelector("textarea"); const resultBox = document.querySelector("#result, .ai-result, [data-result]"); const langSelect = document.querySelector("select"); const basePrompt = promptInput ? promptInput.value.trim() : ""; if (!basePrompt) { if (resultBox) { resultBox.textContent = "કૃપા કરીને પહેલા વિષય અથવા પ્રોમ્પ્ટ લખો."; } return; } const language = langSelect ? langSelect.value : "ગુજરાતી"; const finalPrompt = `${basePrompt}\n\nOutput language: ${language}`; if (resultBox) { resultBox.textContent = "જવાબ તૈયાર થઈ રહ્યો છે..."; } try { const res = await fetch("/api/generate", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ prompt: finalPrompt, model: "gemini-2.5-flash" }) }); const data = await res.json(); if (!res.ok) { throw new Error(data.error || "API માંથી ભૂલ આવી છે."); } const replyText = data?.candidates?.[0]?.content?.parts?.[0]?.text || "કોઈ જવાબ મળ્યો નથી."; if (resultBox) { resultBox.textContent = replyText; } } catch (err) { if (resultBox) { resultBox.textContent = "ભૂલ: " + (err.message || "કનેક્શન સમસ્યા"); } } }
